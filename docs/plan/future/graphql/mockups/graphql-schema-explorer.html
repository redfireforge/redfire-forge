SDL Definition 
"""
Represents a customer order.
Contains order items, shipping information,
payment status, and tracking details.
"""
type Order implements Node & Timestamped {
  # Global unique identifier
  id: ID!

  # Human-readable order number
  orderNumber: String!

  # Line items (paginated)
  items(
    first: Int = 10
    after: String
  ): [OrderItem!]!

  customer: User!
  shippingAddress: Address
  createdAt: DateTime!

  legacyId: Int @deprecated(
    reason: "Use id instead"
  )

  notes: String
}
