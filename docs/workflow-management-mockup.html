CLI Equivalents (for Developers)
# Export single workflow $ redfireforge export "Auth Handler" -o auth-handler.json # Export bundle with sub-workflows $ redfireforge export "Order Pipeline" --bundle -o order-bundle.json # Export all in a folder $ redfireforge export --folder "Shared" -o shared.json # Export by tag $ redfireforge export --tag "production" -o prod.json # Import with auto-copy on conflict $ redfireforge import bundle.json --on-conflict=copy # Import replacing conflicts (CI/CD) $ redfireforge import bundle.json --on-conflict=replace # Dry run $ redfireforge import bundle.json --dry-run # List workflows $ redfireforge list --folder "Shared" --format table
